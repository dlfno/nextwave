Here is a Python script using the built-in `csv` and `collections` modules to parse your sales data and summarize total revenue by region. I've designed it to be lightweight and not require external dependencies like Pandas for this specific task.

python 
import csv
from collections import defaultdict

def summarize_revenue_by_region(csv_filepath):
    revenue_summary = defaultdict(float)
    
    try:
        with open(csv_filepath, mode='r', encoding='utf-8') as file:
            reader = csv.DictReader(file)
            
            for row in reader:
                region = row.get('Region')
                # Safely parse revenue, ignoring rows with missing/invalid data
                try:
                    revenue = float(row.get('Revenue', '0').replace(',', ''))
                    if region:
                        revenue_summary[region] += revenue
                except ValueError:
                    continue
                    
    except FileNotFoundError:
        print(f"Error: File not found at {csv_filepath}")
        return None
        
    return dict(revenue_summary)
